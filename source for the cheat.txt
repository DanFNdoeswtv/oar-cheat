namespace paths {
    inline std::string g_data_dir;
    inline bool        g_resolved = false;
    inline bool dir_exists(const char* p) {
        DWORD a = GetFileAttributesA(p);
        return a != INVALID_FILE_ATTRIBUTES && (a & FILE_ATTRIBUTE_DIRECTORY);
    }
    inline bool ensure_dir(const std::string& p) {
        if (p.empty()) return false;
        if (dir_exists(p.c_str())) return true;
        CreateDirectoryA(p.c_str(), nullptr);
        return dir_exists(p.c_str());
    }
    inline std::string folder_name() {
        char user[64]{};
        if (!GetEnvironmentVariableA("USERNAME", user, sizeof(user))) strcpy_s(user, "user");
        return std::string("oar femboy hacks - ") + user;
    }
    inline const std::string& data_dir() {
        if (g_resolved) return g_data_dir;
        g_resolved = true;
        const std::string kFolder = folder_name();
        char buf[MAX_PATH]{};
        { std::string p = std::string("C:\\") + kFolder; if (ensure_dir(p)) { g_data_dir = p; return g_data_dir; } }
        if (GetTempPathA(MAX_PATH, buf) > 0 && buf[0]) {
            std::string p = buf;
            if (!p.empty() && p.back() == '\\') p.pop_back();
            p += "\\"; p += kFolder;
            if (ensure_dir(p)) { g_data_dir = p; return g_data_dir; }
        }
        if (GetEnvironmentVariableA("APPDATA", buf, MAX_PATH) > 0 && buf[0]) {
            std::string p = buf; p += "\\"; p += kFolder;
            if (ensure_dir(p)) { g_data_dir = p; return g_data_dir; }
        }
        if (GetEnvironmentVariableA("LOCALAPPDATA", buf, MAX_PATH) > 0 && buf[0]) {
            std::string p = buf; p += "\\"; p += kFolder;
            if (ensure_dir(p)) { g_data_dir = p; return g_data_dir; }
        }
        return g_data_dir;
    }
    inline std::string log_path() {
        const std::string& d = data_dir();
        if (d.empty()) return {};
        return d + "\\oar_debug.txt";
    }
    inline std::string sanitize_name(const std::string& in) {
        std::string s = in.empty() ? "default" : in;
        for (auto& c : s) {
            if (c == '\\' || c == '/' || c == ':' || c == '*' || c == '?' ||
                c == '"' || c == '<' || c == '>' || c == '|' || c == '\n' ||
                c == '\r' || c == '\t') c = '_';
        }
        if (s.size() > 64) s.resize(64);
        return s;
    }
    inline std::string config_path(const std::string& name) {
        const std::string& d = data_dir();
        if (d.empty()) return {};
        return d + "\\" + sanitize_name(name) + ".cfg";
    }
    inline std::string dump_path(const char* name) {
        const std::string& d = data_dir();
        if (d.empty()) return {};
        return d + "\\" + name;
    }
    inline bool write_all(const std::string& path, const std::string& content) {
        if (path.empty()) return false;
        HANDLE h = CreateFileA(path.c_str(), GENERIC_WRITE, 0, nullptr, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, nullptr);
        if (h == INVALID_HANDLE_VALUE) return false;
        const char* p = content.data();
        DWORD rem = (DWORD)content.size();
        while (rem > 0) {
            DWORD w = 0;
            if (!WriteFile(h, p, rem, &w, nullptr) || w == 0) { CloseHandle(h); return false; }
            p += w; rem -= w;
        }
        CloseHandle(h);
        return true;
    }
    inline bool read_all(const std::string& path, std::string& out) {
        out.clear();
        if (path.empty()) return false;
        HANDLE h = CreateFileA(path.c_str(), GENERIC_READ, FILE_SHARE_READ | FILE_SHARE_WRITE,
            nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr);
        if (h == INVALID_HANDLE_VALUE) return false;
        DWORD size = GetFileSize(h, nullptr);
        if (size == INVALID_FILE_SIZE) { CloseHandle(h); return false; }
        out.resize(size);
        DWORD rd = 0;
        if (size > 0) {
            if (!ReadFile(h, out.data(), size, &rd, nullptr)) { CloseHandle(h); out.clear(); return false; }
            out.resize(rd);
        }
        CloseHandle(h);
        return true;
    }

static void open_console() {
    if (!AllocConsole()) return;
    SetConsoleTitleA(CHEAT_NAME);
    FILE* f = nullptr;
    freopen_s(&f, "CONOUT$", "w", stdout);
    freopen_s(&f, "CONOUT$", "w", stderr);
    freopen_s(&f, "CONIN$", "r", stdin);
    HANDLE h = GetStdHandle(STD_OUTPUT_HANDLE);
    SetConsoleTextAttribute(h, FOREGROUND_GREEN | FOREGROUND_INTENSITY);
    std::ios::sync_with_stdio();
}
static void dbg(const char* fmt, ...) {
    char buf[2048];
    va_list ap; va_start(ap, fmt); vsnprintf(buf, sizeof(buf), fmt, ap); va_end(ap);
    const std::string& lp = paths::log_path();
    if (!lp.empty()) {
        HANDLE h = CreateFileA(lp.c_str(), FILE_APPEND_DATA, FILE_SHARE_READ | FILE_SHARE_WRITE,
            nullptr, OPEN_ALWAYS, FILE_ATTRIBUTE_NORMAL, nullptr);
        if (h != INVALID_HANDLE_VALUE) {
            DWORD w = 0;
            WriteFile(h, buf, (DWORD)strlen(buf), &w, nullptr);
            WriteFile(h, "\r\n", 2, &w, nullptr);
            CloseHandle(h);
        }
    }
    OutputDebugStringA(buf); OutputDebugStringA("\n");
    HANDLE out = GetStdHandle(STD_OUTPUT_HANDLE);
    if (out && out != INVALID_HANDLE_VALUE) {
        DWORD w = 0;
        WriteFile(out, buf, (DWORD)strlen(buf), &w, nullptr);
        WriteFile(out, "\r\n", 2, &w, nullptr);
    }
}
static void dbg_to(const std::string& path, const char* fmt, ...) {
    if (path.empty()) return;
    char buf[2048];
    va_list ap; va_start(ap, fmt); vsnprintf(buf, sizeof(buf), fmt, ap); va_end(ap);
    HANDLE h = CreateFileA(path.c_str(), FILE_APPEND_DATA, FILE_SHARE_READ | FILE_SHARE_WRITE,
        nullptr, OPEN_ALWAYS, FILE_ATTRIBUTE_NORMAL, nullptr);
    if (h == INVALID_HANDLE_VALUE) return;
    DWORD w = 0;
    WriteFile(h, buf, (DWORD)strlen(buf), &w, nullptr);
    WriteFile(h, "\r\n", 2, &w, nullptr);
    CloseHandle(h);
    OutputDebugStringA(buf); OutputDebugStringA("\n");
}

    if (path.empty()) return;
    HANDLE h = CreateFileA(path.c_str(), GENERIC_WRITE, 0, nullptr, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, nullptr);

}
static bool copy_to_clipboard(const char* text) {
    if (!text) return false;
    size_t len = strlen(text) + 1;
    if (!OpenClipboard(nullptr)) return false;
    EmptyClipboard();
    HGLOBAL hMem = GlobalAlloc(GMEM_MOVEABLE, len);
    if (!hMem) { CloseClipboard(); return false; }
    void* dst = GlobalLock(hMem);
    if (!dst) { GlobalFree(hMem); CloseClipboard(); return false; }
    memcpy(dst, text, len);
    GlobalUnlock(hMem);
    SetClipboardData(CF_TEXT, hMem);
    CloseClipboard();
    return true;
}

static bool smemcpy(void* dst, const void* src, size_t n) {
    __try { memcpy(dst, src, n); return true; }
    __except (EXCEPTION_EXECUTE_HANDLER) { return false; }
}
typedef void (*seh_fn_t)();
__declspec(noinline) static bool seh_guard(seh_fn_t fn) {
    __try { fn(); return true; }
    __except (EXCEPTION_EXECUTE_HANDLER) { return false; }

inline bool sane_ptr(const void* p) {
    auto v = reinterpret_cast<uintptr_t>(p);
    if (v == 0 || v < 0x10000 || v > 0x7FFF'FFFF'0000ULL) return false;
    return true;
}
static uintptr_t safe_read_uintptr(void* base, uintptr_t off) {
    if (!sane_ptr(base)) return 0;
    uintptr_t v = 0;
    if (!smemcpy(&v, reinterpret_cast<uint8_t*>(base) + off, sizeof(v))) return 0;
    if (!sane_ptr(reinterpret_cast<void*>(v))) return 0;
    return v;
}
static int32_t safe_read_i32(void* base, uintptr_t off) {
    if (!sane_ptr(base)) return 0;
    int32_t v = 0; smemcpy(&v, reinterpret_cast<uint8_t*>(base) + off, sizeof(v)); return v;
}

    if (!sane_ptr(base)) return 0.f;
    float v = 0.f; smemcpy(&v, reinterpret_cast<uint8_t*>(base) + off, sizeof(v)); return v;
}

    namespace name_cache {
        struct Entry { int32_t idx = 0; uint32_t num = 0; std::string str; };
        inline Entry g_entries[2048] = {};
        inline int   g_next = 0;
        inline std::string raw_name_to_string(int32_t idx, uint32_t num) {
            if (!g_append_string) return {};
            if (idx <= 0 || idx >= 5'000'000) return {};
            FName fn{ idx, num };
            FString out{};
            g_append_string(&fn, &out);
            if (!out.data || out.num <= 1 || out.num > 4096) return {};
            int len = WideCharToMultiByte(CP_UTF8, 0, out.data, out.num - 1, nullptr, 0, nullptr, nullptr);
            if (len <= 0) return {};
            std::string s(len, 0);
            WideCharToMultiByte(CP_UTF8, 0, out.data, out.num - 1, s.data(), len, nullptr, nullptr);
            return s;
        }
        inline const std::string& get(int32_t idx, uint32_t num) {
            static std::string empty;
            if (idx <= 0 || idx >= 5'000'000) return empty;
            for (int i = 0; i < 2048; ++i) {
                const Entry& e = g_entries[i];
                if (e.idx == idx && e.num == num && !e.str.empty()) return e.str;
            }
            Entry& e = g_entries[g_next];
            g_next = (g_next + 1) % 2048;
            e.idx = idx; e.num = num;
            e.str = raw_name_to_string(idx, num);
            return e.str;
        }
    }
    inline std::string name_to_string(FName name) {
        return name_cache::raw_name_to_string(name.comparison_index, name.number);
    }
    inline const std::string& name_to_string_cached(const FName& name) {
        return name_cache::get(name.comparison_index, name.number);
    }
    inline const std::string& cached_name_of(UObject* o) {
        static std::string empty;
        if (!sane_ptr(o)) return empty;
        FName fn{};
        if (!smemcpy(&fn, reinterpret_cast<uint8_t*>(o) + off::UObject_Name, sizeof(fn))) return empty;
        return name_cache::get(fn.comparison_index, fn.number);
    }
    inline std::string class_name_of(UObject* o) {
        if (!sane_ptr(o)) return {};
        uintptr_t cls = safe_read_uintptr(o, off::UObject_Class);
        if (!sane_ptr(reinterpret_cast<void*>(cls))) return {};
        FName n{};
        if (!smemcpy(&n, reinterpret_cast<void*>(cls + off::UObject_Name), sizeof(n))) return {};
        return name_cache::raw_name_to_string(n.comparison_index, n.number);
    }
    // matches Class, BlueprintGeneratedClass, WidgetBlueprintGeneratedClass, etc.
    inline bool is_class_object(UObject* o) {
        std::string cn = class_name_of(o);
        if (cn.empty()) return false;
        return cn.find("Class") != std::string::npos;
    }
    template<typename Fn>
    inline void for_each_object(Fn cb) {
        uintptr_t arr = base + off::GObjects;
        uintptr_t cptr = safe_read_uintptr(reinterpret_cast<void*>(arr), 0);
        if (!sane_ptr(reinterpret_cast<void*>(cptr))) return;
        int32_t num = safe_read_i32(reinterpret_cast<void*>(arr), 0x14);
        int32_t chunks = safe_read_i32(reinterpret_cast<void*>(arr), 0x1C);
        if (num <= 0 || chunks <= 0 || num > 5'000'000) return;
        const int32_t kChunkSize = 0x10000;
        const uintptr_t kStride = 0x18;
        for (int32_t ci = 0; ci < chunks; ++ci) {
            uintptr_t chunk = safe_read_uintptr(reinterpret_cast<void*>(cptr), ci * 8);
            if (!sane_ptr(reinterpret_cast<void*>(chunk))) continue;
            int32_t in_chunk = kChunkSize;
            int32_t rem = num - (ci * kChunkSize);
            if (rem < in_chunk) in_chunk = rem;
            if (in_chunk <= 0) break;
            for (int32_t ii = 0; ii < in_chunk; ++ii) {
                uintptr_t item = chunk + (kStride * ii);
                uintptr_t obj = safe_read_uintptr(reinterpret_cast<void*>(item), 0);
                if (!sane_ptr(reinterpret_cast<void*>(obj))) continue;
                auto* o = reinterpret_cast<UObject*>(obj);
                if (!cb(o)) return;
            }
        }
    }

    inline ue4::UObject* local_player() {
        auto* gi = game_instance(); if (!gi) return nullptr;
        uintptr_t arr_addr = reinterpret_cast<uintptr_t>(gi) + off::UGameInstance_LocalPlayers;
        int32_t n = safe_read_i32(reinterpret_cast<void*>(arr_addr), 8);
        if (n <= 0 || n > 16) return nullptr;
        uintptr_t data = safe_read_uintptr(reinterpret_cast<void*>(arr_addr), 0);
        if (!sane_ptr(reinterpret_cast<void*>(data))) return nullptr;
        uintptr_t first = safe_read_uintptr(reinterpret_cast<void*>(data), 0);
        return sane_ptr(reinterpret_cast<void*>(first)) ? reinterpret_cast<ue4::UObject*>(first) : nullptr;
    }

        check_world();

    }

        }
        if (cached_pawn && !object_alive(cached_pawn)) cached_pawn = nullptr;
        return cached_pawn;
    }
    inline ue4::UObject* g_los_fn = nullptr;
    inline int g_los_state = 0;
    inline ue4::UObject* find_function_on_class(ue4::UObject* cls, const char* name) {
        if (!sane_ptr(cls)) return nullptr;
        uintptr_t child = safe_read_uintptr(cls, off::UStruct_Children);
        int count = 0;
        while (child && count++ < 5000) {
            ue4::FName n{};
            if (!smemcpy(&n, reinterpret_cast<void*>(child + off::UObject_Name), sizeof(n))) break;
            const std::string& s = ue4::name_to_string_cached(n);
            if (!s.empty() && s == name) return reinterpret_cast<ue4::UObject*>(child);
            child = safe_read_uintptr(reinterpret_cast<void*>(child), off::UField_Next);
        }
        return nullptr;
    }
    inline ue4::UObject* resolve_on_chain(ue4::UObject* inst, const char* name) {
        if (!sane_ptr(inst)) return nullptr;
        uintptr_t cls = safe_read_uintptr(inst, off::UObject_Class);
        int d = 0;
        while (cls && d++ < 64) {
            auto* f = find_function_on_class(reinterpret_cast<ue4::UObject*>(cls), name);
            if (f) return f;
            cls = safe_read_uintptr(reinterpret_cast<void*>(cls), off::UStruct_SuperStruct);
        }
        return nullptr;
    }
    inline void init_los() {
        if (g_los_state == 2 || g_los_state == 1) return;
        auto* pc = local_pc();
        if (!pc) return;
        uintptr_t cls = safe_read_uintptr(pc, off::UObject_Class);
        int depth = 0;
        while (cls && depth++ < 32) {
            auto* fn = find_function_on_class(reinterpret_cast<ue4::UObject*>(cls), "LineOfSightTo");
            if (fn) { g_los_fn = fn; g_los_state = 2; return; }
            cls = safe_read_uintptr(reinterpret_cast<void*>(cls), off::UStruct_SuperStruct);
        }
        g_los_state = 1;
    }
    struct LosParams { ue4::UObject* Other; float ViewX, ViewY, ViewZ; bool bAlternateChecks; bool ReturnValue; uint8_t _pad[2]; };
    inline bool has_line_of_sight(ue4::UObject* pc, ue4::UObject* target, float cx, float cy, float cz) {
        if (!pc || !target) return false;
        if (!g_los_fn || !ue4::g_process_event) return true;
        LosParams p{};
        p.Other = target; p.ViewX = cx; p.ViewY = cy; p.ViewZ = cz;
        __try { ue4::g_process_event(pc, g_los_fn, &p); }
        __except (EXCEPTION_EXECUTE_HANDLER) { return false; }

            int i = find(a);
            if (i < 0) { i = g_next; g_next = (g_next + 1) % 128; g_entries[i].actor = a; }
            g_entries[i].has_los = v; g_entries[i].frame = g_frame;
        }
    }

        los_cache::put(target, r);
        return r;
    }

    inline ue4::TArray<ue4::UObject*>* get_actors() {
        auto* w = chain::world();
        if (!w) return nullptr;
        uintptr_t lvl = safe_read_uintptr(w, off::UWorld_PersistentLevel);
        if (!sane_ptr(reinterpret_cast<void*>(lvl))) return nullptr;
        static uintptr_t g_good_off = 0;
        static const uintptr_t candidates[] = { 0x98, 0x28, 0x38, 0xA0, 0x90 };
        if (g_good_off) {
            uintptr_t arr_addr = lvl + g_good_off;
            int32_t n = safe_read_i32(reinterpret_cast<void*>(arr_addr), 8);
            if (n > 0 && n < 100000) {
                uintptr_t data = safe_read_uintptr(reinterpret_cast<void*>(arr_addr), 0);
                if (sane_ptr(reinterpret_cast<void*>(data)))
                    return reinterpret_cast<ue4::TArray<ue4::UObject*>*>(arr_addr);
            }
            g_good_off = 0;
        }
        for (uintptr_t o : candidates) {
            uintptr_t arr_addr = lvl + o;
            int32_t n = safe_read_i32(reinterpret_cast<void*>(arr_addr), 8);
            if (n < 10 || n > 100000) continue;
            uintptr_t data = safe_read_uintptr(reinterpret_cast<void*>(arr_addr), 0);
            if (!sane_ptr(reinterpret_cast<void*>(data))) continue;
            uintptr_t first = safe_read_uintptr(reinterpret_cast<void*>(data), 0);
            if (!sane_ptr(reinterpret_cast<void*>(first))) continue;
            g_good_off = o;
            return reinterpret_cast<ue4::TArray<ue4::UObject*>*>(arr_addr);
        }
        return nullptr;
    }

        return result;
    }

    inline bool is_dead(ue4::UObject* actor) {
        using namespace dead_track;
        if (!sane_ptr(actor)) return false;
        Entry* e = find(actor);
        if (!e) {
            e = alloc(actor);
            int32_t hp = safe_read_i32(actor, g_dead_health_off);
            if (hp < 0 || hp >= 100'000) e->state = 2;
            else if (hp == 0) e->state = 0;
            else { e->state = 1; e->last_hp = hp; }
            return false;
        }
        e->last_seen = g_frame;
        int32_t hp = safe_read_i32(actor, g_dead_health_off);
        if (e->state == 3) {
            if (hp > 0 && hp < 100'000) { e->state = 1; e->last_hp = hp; return false; }
            return true;
        }
        if (e->state == 2) return false;
        if (hp < 0 || hp >= 100'000) { e->state = 2; return false; }
        if (hp > 0) { e->state = 1; e->last_hp = hp; return false; }
        if (e->state == 1 && e->last_hp > 0) { e->state = 3; return true; }
        e->last_hp = 0;
        return false;
    }

    inline bool object_location(ue4::UObject* obj, Vec3& out) {
        out = Vec3{};

        uintptr_t root = safe_read_uintptr(obj, off::AActor_RootComponent);
        if (!sane_ptr(reinterpret_cast<void*>(root))) return false;
        const uintptr_t candidates[] = { off::USceneComponent_RelativeLocation, 0x128, 0x130, 0x120, 0x118, 0x140 };
        for (uintptr_t o : candidates) {
            Vec3 t{};
            if (!smemcpy(&t, reinterpret_cast<void*>(root + o), sizeof(t))) continue;
            if (!std::isfinite(t.x) || !std::isfinite(t.y) || !std::isfinite(t.z)) continue;
            if (fabsf(t.x) > 1e7f || fabsf(t.y) > 1e7f || fabsf(t.z) > 1e7f) continue;
            if (t.x == 0.f && t.y == 0.f && t.z == 0.f) continue;
            out = t; return true;
        }

    }

    inline bool world_to_screen(const Vec3& world, Vec3& screen, int sw, int sh,
        const Vec3& cam_loc, const Rot& cam_rot, float fov_deg) {
        const float d2r = 3.14159265f / 180.f;
        float cp = cosf(cam_rot.pitch * d2r), sp = sinf(cam_rot.pitch * d2r);
        float cy = cosf(cam_rot.yaw * d2r), sy = sinf(cam_rot.yaw * d2r);
        float fx = cp * cy, fy = cp * sy, fz = sp;
        float rx = -sy, ry = cy, rz = 0.f;
        float ux = -sp * cy, uy = -sp * sy, uz = cp;
        float dx = world.x - cam_loc.x, dy = world.y - cam_loc.y, dz = world.z - cam_loc.z;
        float x_cam = dx * rx + dy * ry + dz * rz;
        float y_cam = dx * ux + dy * uy + dz * uz;
        float z_cam = dx * fx + dy * fy + dz * fz;
        if (z_cam < 1.f) return false;
        float tan_half = tanf(fov_deg * d2r * 0.5f);
        float aspect = float(sw) / float(sh);
        float x_ndc = x_cam / (z_cam * tan_half);
        float y_ndc = y_cam * aspect / (z_cam * tan_half);
        screen.x = (sw * 0.5f) * (1.f + x_ndc);
        screen.y = (sh * 0.5f) * (1.f - y_ndc);
        screen.z = z_cam;
        return true;
    }

    inline std::string pretty_name(const std::string& raw) {
        if (raw.empty()) return {};
        std::string s = raw;
        static const char* kPrefixes[] = {
            "SM_Prop_", "SM_Item_", "SM_Tool_", "SM_Chr_", "SM_Bld_", "SM_Env_", "SM_Wep_", "SM_",
            "SK_Wep_", "SK_Chr_", "SK_", "BP_", "ChildActor1_", "ChildActor_"
        };
        for (const char* p : kPrefixes) {
            size_t pl = strlen(p);
            if (s.size() > pl && s.compare(0, pl, p) == 0) { s = s.substr(pl); break; }
        }
        while (true) {
            size_t us = s.rfind('_');
            if (us == std::string::npos || us + 1 >= s.size()) break;
            bool all_digits = true;
            for (size_t i = us + 1; i < s.size(); ++i)
                if (s[i] < '0' || s[i] > '9') { all_digits = false; break; }
            if (!all_digits) break;
            s.resize(us);
        }
        for (auto& c : s) if (c == '_') c = ' ';
        std::string mid; mid.reserve(s.size() + 8);
        for (size_t i = 0; i < s.size(); ++i) {
            char c = s[i];
            if (i > 0 && c >= 'A' && c <= 'Z') {
                char prev = s[i - 1];
                bool pl = (prev >= 'a' && prev <= 'z');
                bool pu = (prev >= 'A' && prev <= 'Z');
                bool nl = (i + 1 < s.size() && s[i + 1] >= 'a' && s[i + 1] <= 'z');
                if (pl || (pu && nl)) mid.push_back(' ');
            }
            mid.push_back(c);
        }
        std::string out; out.reserve(mid.size());
        bool last_space = true;
        for (char c : mid) {
            if (c == ' ') { if (!last_space) { out.push_back(' '); last_space = true; } }
            else { out.push_back(c); last_space = false; }
        }
        while (!out.empty() && out.back() == ' ') out.pop_back();
        for (auto& c : out) if (c >= 'a' && c <= 'z') c -= 32;
        return out.empty() ? raw : out;
    }

    inline void draw() {
        if (!g_on) return;
        if (!chain::world()) return;
        esp::Vec3 cam_loc{}; esp::Rot cam_rot{}; float cam_fov = 0.f;
        if (!esp::read_camera(cam_loc, cam_rot, cam_fov)) return;
        auto* arr = actors::get_actors();
        if (!arr) return;
        uintptr_t arr_addr = reinterpret_cast<uintptr_t>(arr);
        int32_t n = safe_read_i32(reinterpret_cast<void*>(arr_addr), 8);
        if (n <= 0 || n > 100000) return;
        uintptr_t data = safe_read_uintptr(reinterpret_cast<void*>(arr_addr), 0);
        if (!sane_ptr(reinterpret_cast<void*>(data))) return;
        ImDrawList* dl = ImGui::GetBackgroundDrawList();
        const float cx = g_screen_w * 0.5f;
        const float cy = g_screen_h * 0.5f;
        for (int i = 0; i < n; ++i) {
            uintptr_t actor_ptr = safe_read_uintptr(reinterpret_cast<void*>(data), i * 8);
            if (!sane_ptr(reinterpret_cast<void*>(actor_ptr))) continue;
            auto* actor = reinterpret_cast<ue4::UObject*>(actor_ptr);
            const std::string& name = ue4::cached_name_of(actor);
            if (name.empty()) continue;
            Classified c = classify(name);
            if (c.cat == ItemCat::Skip) continue;
            if (!cat_enabled(c.cat)) continue;
            if (esp::is_pawn_class(actor)) {
                if (c.cat != ItemCat::Camera && c.cat != ItemCat::Vault &&
                    c.cat != ItemCat::Utility && c.cat != ItemCat::SecurityRoom) continue;
            }
            esp::Vec3 wp{};
            if (!esp::object_location(actor, wp)) continue;
            float dx = wp.x - cam_loc.x, dy = wp.y - cam_loc.y, dz = wp.z - cam_loc.z;
            float dist_m = sqrtf(dx * dx + dy * dy + dz * dz) / 100.f;
            if (dist_m > g_max_dist) continue;

            if (!esp::on_screen(scr, 200.f, 200.f)) continue;
            dl->AddCircle({ scr.x, scr.y }, 7.f, c.color, 16, 1.5f);
            dl->AddCircleFilled({ scr.x, scr.y }, 2.5f, c.color, 6);
            char lbl[128];
            snprintf(lbl, sizeof(lbl), "%s  %.0fm", c.label, dist_m);
            ImVec2 sz = ImGui::CalcTextSize(lbl);
            ImVec2 tp = { scr.x - sz.x * 0.5f, scr.y + 10.f };
            dl->AddText({ tp.x + 1.f, tp.y + 1.f }, IM_COL32(0, 0, 0, 220), lbl);
            dl->AddText(tp, c.color, lbl);
            if (g_show_names) {
                std::string display = pretty_name(name);
                if (!display.empty()) {
                    ImVec2 nsz = ImGui::CalcTextSize(display.c_str());
                    ImVec2 np = { scr.x - nsz.x * 0.5f, tp.y + sz.y + 1.f };

                    dl->AddText(np, IM_COL32(230, 230, 230, 240), display.c_str());
                }
            }
            if (g_draw_line) dl->AddLine({ cx, cy }, { scr.x, scr.y }, (c.color & 0x00FFFFFF) | 0x66000000, 1.0f);
        }
    }

namespace aim {
    struct Rot { float pitch, yaw, roll; };
    inline Rot get_control_rotation() {
        auto* pc = chain::local_pc(); Rot r{};
        if (!pc) return r;
        if (!smemcpy(&r, reinterpret_cast<uint8_t*>(pc) + off::APlayerController_ControlRotation, sizeof(r))) return Rot{};
        if (!std::isfinite(r.pitch) || !std::isfinite(r.yaw)) return Rot{};
        return r;
    }
    inline bool set_control_rotation(const Rot& r) {
        auto* pc = chain::local_pc();
        if (!pc) return false;
        return smemcpy(reinterpret_cast<uint8_t*>(pc) + off::APlayerController_ControlRotation, &r, sizeof(r));
    }
    inline Rot look_at(const esp::Vec3& from, const esp::Vec3& to) {
        Rot r{};
        float dx = to.x - from.x, dy = to.y - from.y, dz = to.z - from.z;
        float d2d = sqrtf(dx * dx + dy * dy);
        r.pitch = atan2f(dz, d2d) * 180.f / 3.14159265f;
        r.yaw = atan2f(dy, dx) * 180.f / 3.14159265f;
        r.roll = 0.f;
        return r;
    }
    inline float norm_deg(float a) { while (a > 180.f) a -= 360.f; while (a < -180.f) a += 360.f; return a; }
    inline Rot lerp(const Rot& a, const Rot& b, float t) {
        Rot r{};
        r.pitch = a.pitch + norm_deg(b.pitch - a.pitch) * t;
        r.yaw = a.yaw + norm_deg(b.yaw - a.yaw) * t;
        return r;
    }
} // namespace aim

    struct SetBoolParams { bool v; uint8_t pad[7]; };
    struct SetIntParams { int32_t v; uint8_t pad[4]; };
    inline uintptr_t find_mesh_off(ue4::UObject* actor) {
        if (!sane_ptr(actor)) return 0;
        static const uintptr_t kCands[] = { 0x290, 0x2A0, 0x298, 0x288, 0x280, 0x2A8, 0x2B0, 0x2B8, 0x2C0 };
        for (uintptr_t o : kCands) {
            uintptr_t cand = safe_read_uintptr(actor, o);
            if (!sane_ptr(reinterpret_cast<void*>(cand))) continue;
            std::string cn = ue4::class_name_of(reinterpret_cast<ue4::UObject*>(cand));
            if (cn.find("SkeletalMesh") != std::string::npos ||
                cn.find("StaticMesh") != std::string::npos ||
                cn.find("MeshComponent") != std::string::npos) return o;
        }
        return 0;
    }
    inline void resolve() {
        if (g_set_depth_state != 0 && g_set_stencil_state != 0 && g_mesh_off_resolved) return;
        auto* pawn = chain::local_pawn();
        if (!pawn) return;
        if (!g_mesh_off_resolved) {
            uintptr_t off = find_mesh_off(pawn);
            if (off) { g_mesh_off = off; g_mesh_off_resolved = true; off::ACharacter_Mesh = off; }
        }
        uintptr_t mesh = safe_read_uintptr(pawn, g_mesh_off);
        if (!sane_ptr(reinterpret_cast<void*>(mesh))) return;
        if (g_set_depth_state == 0) {
            g_set_depth_fn = chain::resolve_on_chain(reinterpret_cast<ue4::UObject*>(mesh), "SetRenderCustomDepth");
            if (!g_set_depth_fn) g_set_depth_fn = chain::resolve_on_chain(reinterpret_cast<ue4::UObject*>(mesh), "SetRenderCustomDepthEnabled");
            g_set_depth_state = g_set_depth_fn ? 2 : 1;
        }
        if (g_set_stencil_state == 0) {
            g_set_stencil_fn = chain::resolve_on_chain(reinterpret_cast<ue4::UObject*>(mesh), "SetCustomDepthStencilValue");
            g_set_stencil_state = g_set_stencil_fn ? 2 : 1;
        }
    }
    inline void apply_to(ue4::UObject* actor, bool visible) {
        if (!g_set_depth_fn || !ue4::g_process_event) return;
        uintptr_t mesh = safe_read_uintptr(actor, g_mesh_off);
        if (!sane_ptr(reinterpret_cast<void*>(mesh))) return;
        SetBoolParams p{}; p.v = true;
        __try { ue4::g_process_event(reinterpret_cast<ue4::UObject*>(mesh), g_set_depth_fn, &p); }
        __except (EXCEPTION_EXECUTE_HANDLER) {}
        if (g_set_stencil_fn) {
            SetIntParams sp{}; sp.v = visible ? g_stencil : (g_stencil + 1);
            __try { ue4::g_process_event(reinterpret_cast<ue4::UObject*>(mesh), g_set_stencil_fn, &sp); }
            __except (EXCEPTION_EXECUTE_HANDLER) {}
        }
    }

    inline void scan_weapons() {
        g_weapon_count = 0;
        ue4::for_each_object([](ue4::UObject* o) -> bool {
            if (g_weapon_count >= 128) return false;
            if (!ue4::is_class_object(o)) return true;
            const std::string& nm = ue4::cached_name_of(o);
            if (nm.size() < 3 || nm.substr(nm.size() - 2) != "_C") return true;
            if (nm.find("_GEN_VARIABLE_") != std::string::npos) return true;
            if (nm.find("Attachment") != std::string::npos) return true;
            if (nm.find("GunBase") != std::string::npos) return true;
            if (nm.find("Item") != std::string::npos) return true;
            if (nm.find("AttachedBackGun") != std::string::npos) return true;
            std::string pretty = pretty_weapon_name(nm);
            if (pretty.empty()) return true;
            for (int j = 0; j < g_weapon_count; ++j)
                if (g_weapons[j].display_name == pretty) return true;
            g_weapons[g_weapon_count].weapon_class = o;
            g_weapons[g_weapon_count].raw_name = nm;
            g_weapons[g_weapon_count].display_name = pretty;
            g_weapon_count++;
            return true;
            });
        g_weapons_scanned = true;
        dbg("skins: scanned %d weapon classes", g_weapon_count);
    }

    inline void scan_assets() {
        g_mat_count = 0;
        ue4::for_each_object([](ue4::UObject* o) -> bool {
            std::string cls = ue4::class_name_of(o);
            if (cls.empty()) return true;
            if (cls == "Material" || cls == "MaterialInstance" ||
                cls == "MaterialInstanceConstant" || cls == "MaterialInstanceDynamic") {
                if (g_mat_count < 2048) {
                    g_mat_assets[g_mat_count].obj = o;
                    g_mat_assets[g_mat_count].name = ue4::cached_name_of(o);
                    g_mat_assets[g_mat_count].cls = cls;
                    g_mat_count++;
                }
            }
            return true;
            });
        g_scanned = true;
        dbg("skins: scanned %d materials", g_mat_count);
    }

    inline uintptr_t find_mesh_comp(ue4::UObject* actor) {
        if (!sane_ptr(actor)) return 0;
        static const uintptr_t kCands[] = { 0x290, 0x2A0, 0x298, 0x288, 0x280, 0x2A8, 0x2B0, 0x2B8, 0x2C0, 0x2C8 };
        for (uintptr_t o : kCands) {

    inline void resolve() {
        auto* pawn = chain::local_pawn();
        if (pawn) { uintptr_t m = find_mesh_comp(pawn); if (m) g_pawn_mesh = m; }

            g_set_material_fn = chain::resolve_on_chain(reinterpret_cast<ue4::UObject*>(any), "SetMaterial");
            g_set_mat_state = g_set_material_fn ? 2 : 1;
        }
    }

        p.element_index = slot;
        p.material = mat;
        __try { ue4::g_process_event(reinterpret_cast<ue4::UObject*>(mesh), g_set_material_fn, &p); }
        __except (EXCEPTION_EXECUTE_HANDLER) {}
    }

        if (!g_weapon_mesh) { dbg("skins: no weapon mesh"); return; }

    inline ue4::UObject* find_weapon_instance(ue4::UObject* weapon_class) {
        if (!sane_ptr(weapon_class)) return nullptr;
        ue4::UObject* found = nullptr;
        uintptr_t target = reinterpret_cast<uintptr_t>(weapon_class);
        ue4::for_each_object([&](ue4::UObject* o) -> bool {
            if (o == weapon_class) return true;
            uintptr_t cls = safe_read_uintptr(o, off::UObject_Class);
            if (cls != target) return true;
            found = o;
            return false;
            });
        return found;
    }

    inline void apply_material_to_selected_weapon() {
        if (g_weapon_sel < 0 || g_weapon_sel >= g_weapon_count) return;
        if (g_mat_sel < 0 || g_mat_sel >= g_mat_count) return;
        auto* wc = g_weapons[g_weapon_sel].weapon_class;
        auto* inst = find_weapon_instance(wc);
        if (!inst) { dbg("skins: no instance of %s", g_weapons[g_weapon_sel].display_name.c_str()); return; }
        uintptr_t mesh = find_mesh_comp(inst);
        if (!mesh) { dbg("skins: no mesh on weapon"); return; }
        apply_material(mesh, g_mat_assets[g_mat_sel].obj, 0);
        dbg("skins: %s -> %s", g_mat_assets[g_mat_sel].name.c_str(),
            g_weapons[g_weapon_sel].display_name.c_str());
    }

    inline void tick_reapply() {
        if (!ue4::g_process_event) return;
        auto* w = weapon::local_weapon();
        uintptr_t wm = w ? find_mesh_comp(w) : 0;
        if (wm) {
            g_weapon_mesh = wm;
            if (wm != g_last_weapon_mesh_ptr) {
                g_last_weapon_mesh_ptr = wm;
                if (g_persistent_weapon_mat && g_set_material_fn)
                    apply_material(wm, g_persistent_weapon_mat, 0);
            }
        }
        auto* pawn = chain::local_pawn();
        uintptr_t pm = pawn ? find_mesh_comp(pawn) : 0;
        if (pm) {
            g_pawn_mesh = pm;
            if (pm != g_last_pawn_mesh_ptr) {
                g_last_pawn_mesh_ptr = pm;
                if (g_persistent_pawn_mat && g_set_material_fn)
                    apply_material(pm, g_persistent_pawn_mat, 0);
            }
        }
    }

        g_last_pawn_mesh_ptr = 0;

    inline bool name_looks_like_sky(const std::string& n) {
        if (n.find("Sky") != std::string::npos) return true;
        if (n.find("Atmosphere") != std::string::npos) return true;
        if (n.find("Cloud") != std::string::npos) return true;
        if (n.find("Sunset") != std::string::npos) return true;
        if (n.find("Dawn") != std::string::npos) return true;
        if (n.find("Dusk") != std::string::npos) return true;
        return false;
    }
    inline ue4::UObject* find_sky_actor() {
        if (g_sky_actor && chain::object_alive(g_sky_actor)) return g_sky_actor;
        g_sky_actor = nullptr;
        auto* arr = actors::get_actors();
        if (arr) {
            uintptr_t arr_addr = reinterpret_cast<uintptr_t>(arr);
            int32_t n = safe_read_i32(reinterpret_cast<void*>(arr_addr), 8);
            uintptr_t data = safe_read_uintptr(reinterpret_cast<void*>(arr_addr), 0);
            if (sane_ptr(reinterpret_cast<void*>(data)) && n > 0 && n < 100000) {
                for (int i = 0; i < n; ++i) {
                    uintptr_t ap = safe_read_uintptr(reinterpret_cast<void*>(data), i * 8);
                    if (!sane_ptr(reinterpret_cast<void*>(ap))) continue;
                    auto* actor = reinterpret_cast<ue4::UObject*>(ap);
                    const std::string& cn = ue4::class_name_of(actor);
                    if (cn.find("Character") != std::string::npos) continue;
                    if (cn.find("Pawn") != std::string::npos) continue;
                    if (name_looks_like_sky(cn)) {
                        g_sky_actor = actor;
                        dbg("skybox: actor = %s @ %p", cn.c_str(), actor);
                        break;
                    }
                }
            }
        }
        if (!g_sky_actor) {
            ue4::for_each_object([](ue4::UObject* o) -> bool {
                const std::string& cn = ue4::class_name_of(o);
                if (cn.empty()) return true;
                if (cn.find("SkySphere") == std::string::npos &&
                    cn.find("SkyDome") == std::string::npos &&
                    cn.find("BP_Sky") == std::string::npos) return true;
                if (cn.size() > 2 && cn.substr(cn.size() - 2) == "_C") return true;
                g_sky_actor = o;
                dbg("skybox: GObjects hit = %s @ %p", cn.c_str(), o);
                return false;
                });
        }
        return g_sky_actor;
    }
    inline uintptr_t find_sky_mesh(ue4::UObject* actor) {
        if (!sane_ptr(actor)) return 0;
        static const uintptr_t kCands[] = {
            0x130, 0x2A0, 0x290, 0x288, 0x280, 0x298, 0x2A8, 0x2B0, 0x2B8, 0x2C0, 0x2C8, 0x138, 0x140
        };
        for (uintptr_t o : kCands) {
            uintptr_t cand = safe_read_uintptr(actor, o);
            if (!sane_ptr(reinterpret_cast<void*>(cand))) continue;
            std::string cn = ue4::class_name_of(reinterpret_cast<ue4::UObject*>(cand));
            if (cn.find("StaticMesh") != std::string::npos ||
                cn.find("SkeletalMesh") != std::string::npos ||
                cn.find("MeshComponent") != std::string::npos) return cand;
        }
        return 0;
    }
    inline void resolve() {
        if (g_scan_state == 2) return;
        auto* actor = find_sky_actor();
        if (!actor) { g_scan_state = 1; return; }
        g_sky_mesh = find_sky_mesh(actor);
        if (!g_sky_mesh) { g_scan_state = 1; dbg("skybox: no mesh on sky actor"); return; }
        if (!skins::g_set_material_fn) {
            skins::g_set_material_fn = chain::resolve_on_chain(reinterpret_cast<ue4::UObject*>(g_sky_mesh), "SetMaterial");
            if (skins::g_set_material_fn) skins::g_set_mat_state = 2;
        }
        g_scan_state = 2;
        dbg("skybox: ready (mesh=%p)", (void*)g_sky_mesh);
    }
    inline void apply(ue4::UObject* mat) {
        if (!mat) return;
        if (g_scan_state != 2) resolve();
        if (!g_sky_mesh || !skins::g_set_material_fn || !ue4::g_process_event) return;
        SetMatParams p{};
        p.element_index = g_mat_slot;
        p.material = mat;
        __try { ue4::g_process_event(reinterpret_cast<ue4::UObject*>(g_sky_mesh), skins::g_set_material_fn, &p); }
        __except (EXCEPTION_EXECUTE_HANDLER) { return; }
        g_current = mat;
        dbg("skybox: applied slot %d", g_mat_slot);
    }
    inline void reset() {
        if (g_sky_mesh && skins::g_set_material_fn && ue4::g_process_event) {
            SetMatParams p{};
            p.element_index = g_mat_slot;
            p.material = nullptr;
            __try { ue4::g_process_event(reinterpret_cast<ue4::UObject*>(g_sky_mesh), skins::g_set_material_fn, &p); }
            __except (EXCEPTION_EXECUTE_HANDLER) {}
        }
        g_current = nullptr;
    }
    inline uintptr_t g_last_mesh = 0;
    inline void tick_reapply() {
        if (!g_current) return;
        if (!g_sky_mesh) resolve();
        if (!g_sky_mesh) return;
        if (g_sky_mesh != g_last_mesh) {
            g_last_mesh = g_sky_mesh;
            apply(g_current);
        }
    }
} // namespace skybox

namespace speedhack {
    using QPC_t = BOOL(WINAPI*)(LARGE_INTEGER*);
    inline QPC_t o_qpc = nullptr;
    inline float          g_speed = 1.0f;
    inline bool           g_on = false;
    inline LARGE_INTEGER  g_last_real = {};
    inline LARGE_INTEGER  g_fake = {};
    inline LARGE_INTEGER  g_qpc_freq = {};
    inline CRITICAL_SECTION g_cs;
    inline bool           g_cs_init = false;
    inline bool           g_installed = false;
    BOOL WINAPI hk_qpc(LARGE_INTEGER* out) {
        if (!o_qpc) return FALSE;
        LARGE_INTEGER real;
        if (!o_qpc(&real)) return FALSE;
        if (!g_cs_init) { *out = real; return TRUE; }
        EnterCriticalSection(&g_cs);
        LONGLONG real_delta = real.QuadPart - g_last_real.QuadPart;
        g_last_real = real;
        float sp = g_on ? g_speed : 1.0f;
        if (sp < 0.f) sp = 0.f;
        LONGLONG scaled = (LONGLONG)((double)real_delta * (double)sp);
        g_fake.QuadPart += scaled;
        LARGE_INTEGER result = g_fake;
        LeaveCriticalSection(&g_cs);
        *out = result;
        return TRUE;
    }
    inline void install() {
        if (g_installed) return;
        if (!g_cs_init) { InitializeCriticalSection(&g_cs); g_cs_init = true; }
        QueryPerformanceFrequency(&g_qpc_freq);
        QueryPerformanceCounter(&g_last_real);
        g_fake = g_last_real;
        HMODULE k32 = GetModuleHandleA("kernel32.dll");
        if (!k32) return;
        void* addr = (void*)GetProcAddress(k32, "QueryPerformanceCounter");
        if (!addr) return;
        MH_STATUS st = MH_CreateHook(addr, &hk_qpc, reinterpret_cast<LPVOID*>(&o_qpc));
        if (st != MH_OK) return;
        st = MH_EnableHook(addr);
        if (st != MH_OK) return;
        g_installed = true;
    }
} // namespace speedhack

namespace recoil {
    inline bool     g_on = false;
    inline bool     g_skip_call = false;
    inline int32_t  g_cmp_idx = 0;
    inline bool     g_resolved = false;
    inline void resolve() {
        if (g_resolved) return;
        g_resolved = true;
        ue4::for_each_object([](ue4::UObject* o) -> bool {
            ue4::FName fn{};
            if (!smemcpy(&fn, reinterpret_cast<void*>(reinterpret_cast<uintptr_t>(o) + off::UObject_Name), sizeof(fn))) return true;
            if (fn.comparison_index <= 0) return true;
            const std::string& name = ue4::name_to_string_cached(fn);
            if (name == "AlphaRecoil__UpdateFunc" || name == "Recoil__UpdateFunc") g_cmp_idx = fn.comparison_index;
            return true;
            });
    }
    inline bool intercept(ue4::UObject* fn, void* params) {
        if (!g_on || !fn || !g_cmp_idx) return false;
        ue4::FName fn_name{};
        if (!smemcpy(&fn_name, reinterpret_cast<uint8_t*>(fn) + 0x18, sizeof(fn_name))) return false;
        if (fn_name.comparison_index != g_cmp_idx) return false;
        if (params) { float zero = 0.f; smemcpy(params, &zero, sizeof(zero)); }
        return g_skip_call;
    }
} // namespace recoil

namespace sav_gen {

    inline std::string get_username() {
        if (!g_username.empty()) return g_username;
        char buf[256];
        DWORD n = GetEnvironmentVariableA("USERNAME", buf, sizeof(buf));
        if (n > 0 && n < sizeof(buf)) g_username = buf;
        else g_username = "(unknown)";
        return g_username;
    }

    inline std::string resolve_oar_save_folder() {
        char buf[MAX_PATH];
        if (GetEnvironmentVariableA("LOCALAPPDATA", buf, MAX_PATH) == 0) return {};
        std::string base = buf;
        std::string p1 = base + "\\OAR";
        std::string p2 = p1 + "\\Saved";
        std::string p3 = p2 + "\\SaveGames";
        if (GetFileAttributesA(p1.c_str()) == INVALID_FILE_ATTRIBUTES) {
            if (!CreateDirectoryA(p1.c_str(), nullptr)) return {};
        }
        if (GetFileAttributesA(p2.c_str()) == INVALID_FILE_ATTRIBUTES) {
            if (!CreateDirectoryA(p2.c_str(), nullptr)) return {};
        }
        if (GetFileAttributesA(p3.c_str()) == INVALID_FILE_ATTRIBUTES) {
            if (!CreateDirectoryA(p3.c_str(), nullptr)) return {};
            g_folder_created = true;
        }
        return p3;
    }

    inline std::string find_save_folder() {
        std::string oar = resolve_oar_save_folder();
        if (!oar.empty()) return oar;
        return {};
    }

    inline void scan_folder() {
        g_sav_files.clear();
        get_username();
        g_folder = find_save_folder();
        if (g_folder.empty()) { g_folder_scanned = true; return; }
        std::string pat = g_folder + "\\*.sav";
        WIN32_FIND_DATAA fd;
        HANDLE h = FindFirstFileA(pat.c_str(), &fd);
        if (h != INVALID_HANDLE_VALUE) {
            do { g_sav_files.push_back(g_folder + "\\" + fd.cFileName); } while (FindNextFileA(h, &fd));
            FindClose(h);
        }
        g_sav_verify_count = (int)g_sav_files.size();
        g_folder_scanned = true;
        dbg("sav_gen: user=%s folder=%s files=%d", g_username.c_str(), g_folder.c_str(), g_sav_verify_count);
    }

    // KEY: UnlockedMaps stores ShopItem_Map_*_C classes. Include them.
    inline bool class_name_is_map(const std::string& n) {
        if (n.size() < 3 || n.substr(n.size() - 2) != "_C") return false;
        if (n.find("_GEN_VARIABLE_") != std::string::npos) return false;
        // The shop-items that represent map ownership
        if (n.find("ShopItem_Map_") == 0) {
            if (n == "ShopItem_Map_base_C") return false;   // base class itself
            return true;
        }
        return false;
    }

    inline int scan_maps() {
        g_map_count = 0;
        ue4::for_each_object([](ue4::UObject* o) -> bool {
            if (g_map_count >= 64) return false;
            if (!ue4::is_class_object(o)) return true;
            const std::string& nm = ue4::cached_name_of(o);
            if (!class_name_is_map(nm)) return true;
            bool dup = false;
            for (int j = 0; j < g_map_count; ++j) if (g_maps[j] == o) { dup = true; break; }
            if (!dup) { g_maps[g_map_count++] = o; dbg("sav_gen: map -> %s", nm.c_str()); }
            return true;
            });
        g_maps_scanned = true;
        return g_map_count;
    }

    struct TArrRef { uintptr_t addr; uintptr_t data; int32_t num; int32_t max; };
    inline TArrRef read_tarr(uintptr_t addr) {
        TArrRef r{}; r.addr = addr;
        r.data = safe_read_uintptr(reinterpret_cast<void*>(addr), 0);
        r.num = safe_read_i32(reinterpret_cast<void*>(addr), 8);
        r.max = safe_read_i32(reinterpret_cast<void*>(addr), 12);
        return r;
    }
    inline bool write_tarr_meta(uintptr_t addr, uintptr_t data, int32_t num, int32_t max) {
        if (!smemcpy(reinterpret_cast<uint8_t*>(addr), &data, 8)) return false;
        if (!smemcpy(reinterpret_cast<uint8_t*>(addr) + 8, &num, 4)) return false;
        if (!smemcpy(reinterpret_cast<uint8_t*>(addr) + 12, &max, 4)) return false;
        return true;
    }

    // ownership check via the live UnlockedMaps
    inline bool is_map_owned(ue4::UObject* pc, ue4::UObject* map_class) {
        uintptr_t arr_addr = reinterpret_cast<uintptr_t>(pc) + off::RobberController_UnlockedMaps;
        TArrRef r = read_tarr(arr_addr);
        if (!sane_ptr(reinterpret_cast<void*>(r.data)) || r.num <= 0 || r.num > 256) return false;
        uintptr_t target = reinterpret_cast<uintptr_t>(map_class);
        for (int32_t i = 0; i < r.num; ++i) {
            uintptr_t e = safe_read_uintptr(reinterpret_cast<void*>(r.data), (uintptr_t)i * 8);
            if (e == target) return true;
        }
        return false;
    }

    inline void resolve_addmap() {
        if (g_addmap_state != 0) return;
        auto* pc = chain::local_pc();

        g_addmap_fn = chain::resolve_on_chain(pc, "AddMap");
        g_addmap_state = g_addmap_fn ? 2 : 1;

    inline int inject_maps_via_addmap() {
        auto* pc = chain::local_pc();
        if (!pc) return 0;
        if (!g_maps_scanned) scan_maps();
        if (g_map_count == 0) return 0;
        resolve_addmap();
        if (!g_addmap_fn || !ue4::g_process_event) return 0;
        int added = 0, skipped = 0;
        for (int i = 0; i < g_map_count; ++i) {
            if (is_map_owned(pc, g_maps[i])) { skipped++; continue; }
            AddMapParams p{};
            p.Map = g_maps[i];
            __try {
                ue4::g_process_event(pc, g_addmap_fn, &p);
                added++;
            }
            __except (EXCEPTION_EXECUTE_HANDLER) {}
        }
        dbg("sav_gen: AddMap +%d (skipped %d owned)", added, skipped);
        return added;
    }

    inline void dump_shop_items() {
        const std::string path = paths::dump_path("shop_items.txt");
        dbg_to_begin(path);
        dbg_to(path, "=== ShopItem_Map_* CDO dump ===");
        ue4::for_each_object([&](ue4::UObject* o) -> bool {
            if (!ue4::is_class_object(o)) return true;
            const std::string& nm = ue4::cached_name_of(o);
            if (nm.find("ShopItem_Map_") != 0) return true;
            if (nm.size() < 3 || nm.substr(nm.size() - 2) != "_C") return true;

            dbg_to(path, "\n--- %s @ %p ---", nm.c_str(), (void*)o);

            static const uintptr_t kCdoOffsets[] = { 0x118, 0x110, 0x108, 0x120, 0x0F8, 0x100 };
            uintptr_t cdo = 0;
            for (uintptr_t off : kCdoOffsets) {
                uintptr_t c = safe_read_uintptr(o, off);
                if (!sane_ptr(reinterpret_cast<void*>(c))) continue;
                std::string cn = ue4::class_name_of(reinterpret_cast<ue4::UObject*>(c));
                if (cn == "ShopItem_Map_base_C" || cn.find("ShopItem") != std::string::npos) {
                    cdo = c;
                    dbg_to(path, "  CDO @ off 0x%X = %p (class=%s)", (unsigned)off, (void*)c, cn.c_str());
                    break;
                }
            }
            if (!cdo) { dbg_to(path, "  CDO not found"); return true; }

            dbg_to(path, "  pointer scan:");
            for (uintptr_t off = 0x20; off < 0x400; off += 8) {
                uintptr_t v = safe_read_uintptr(reinterpret_cast<void*>(cdo), off);
                if (!sane_ptr(reinterpret_cast<void*>(v))) continue;
                std::string cn = ue4::class_name_of(reinterpret_cast<ue4::UObject*>(v));
                if (cn.empty()) continue;
                const std::string& on = ue4::cached_name_of(reinterpret_cast<ue4::UObject*>(v));
                dbg_to(path, "    [0x%X] %p  obj=%s  cls=%s",
                    (unsigned)off, (void*)v, on.c_str(), cn.c_str());
            }
            return true;
            });
        dbg_to(path, "\n=== end ===");
        dbg("shop items dump -> %s", path.c_str());
    }

namespace stealth {
    inline bool g_on = false;
    inline const char* g_patterns[] = {
        "OnPlayerSeen", "OnSeePlayer", "OnPlayerDetected", "OnPlayerSpotted",
        "ReportPlayer", "ReportSeen", "DetectPlayer", "SpotPlayer", "SensePlayer",
        "CanSeePlayer", "CanSeePawn", "VisionCheck", "SightCheck",
        "PlayerDetected", "AlertPlayer",
    };
    inline constexpr int g_pattern_count = sizeof(g_patterns) / sizeof(g_patterns[0]);
    inline int32_t g_cmp_idxs[256] = {};
    inline int g_cmp_count = 0;
    inline bool g_resolved = false;
    inline bool g_filter_onseen = true;
    inline bool g_filter_report = true;
    inline bool g_filter_detect = true;
    inline bool g_filter_cansee = true;
    inline bool g_filter_misc = true;
    inline bool pattern_enabled(const char* p) {
        std::string s = p;
        if (s.find("OnPlayerSeen") != std::string::npos || s.find("OnSeePlayer") != std::string::npos ||
            s.find("OnPlayerDetected") != std::string::npos || s.find("OnPlayerSpotted") != std::string::npos ||
            s.find("PlayerDetected") != std::string::npos) return g_filter_onseen;
        if (s.find("Report") != std::string::npos) return g_filter_report;
        if (s.find("Detect") != std::string::npos || s.find("Spot") != std::string::npos) return g_filter_detect;
        if (s.find("CanSee") != std::string::npos) return g_filter_cansee;
        return g_filter_misc;
    }
    inline void resolve() {
        if (g_resolved) return;
        g_resolved = true;
        int found = 0;
        ue4::for_each_object([&](ue4::UObject* o) -> bool {
            if (found >= 256) return false;
            ue4::FName fn{};
            if (!smemcpy(&fn, reinterpret_cast<uint8_t*>(o) + off::UObject_Name, sizeof(fn))) return true;
            if (fn.comparison_index <= 0) return true;
            const std::string& nm = ue4::name_to_string_cached(fn);
            if (nm.empty()) return true;
            for (int p = 0; p < g_pattern_count; ++p) {
                if (nm.find(g_patterns[p]) != std::string::npos) {
                    bool dup = false;
                    for (int k = 0; k < found; ++k) if (g_cmp_idxs[k] == fn.comparison_index) { dup = true; break; }
                    if (!dup) g_cmp_idxs[found++] = fn.comparison_index;
                    break;
                }
            }
            return true;
            });
        g_cmp_count = found;
        dbg("stealth: resolved %d", found);
    }
    inline bool is_target(int32_t cmp_idx) {
        if (!g_on || cmp_idx <= 0 || g_cmp_count <= 0) return false;
        for (int i = 0; i < g_cmp_count; ++i) if (g_cmp_idxs[i] == cmp_idx) return true;
        return false;
    }
    inline void dump_sensing_fns() {
        const std::string path = paths::dump_path("stealth_dump.txt");
        dbg_to_begin(path);
        dbg_to(path, "=== stealth dump ===");
        dbg_to(path, "resolved: %d", g_cmp_count);
        for (int i = 0; i < g_cmp_count; ++i) dbg_to(path, "  cmp_idx=0x%X", g_cmp_idxs[i]);
        dbg_to(path, "=== end ===");
        dbg("stealth dump -> %s", path.c_str());
    }
} // namespace stealth

    struct PickResult { esp::Vec3 pos{}; ue4::UObject* actor = nullptr; bool found = false; };
    inline PickResult pick_target() {
        PickResult r{};
        if (!chain::world()) return r;
        esp::Vec3 cam_loc{}; esp::Rot cam_rot{}; float cam_fov = 0.f;
        if (!esp::read_camera(cam_loc, cam_rot, cam_fov)) return r;
        auto* me_pawn = chain::local_pawn();
        auto* pc = chain::local_pc();
        auto* arr = actors::get_actors();
        if (!arr) return r;
        uintptr_t arr_addr = reinterpret_cast<uintptr_t>(arr);
        int32_t n = safe_read_i32(reinterpret_cast<void*>(arr_addr), 8);
        uintptr_t data = safe_read_uintptr(reinterpret_cast<void*>(arr_addr), 0);
        float cx = g_screen_w * 0.5f, cy = g_screen_h * 0.5f;
        float fov_px = (g_aim_fov / 90.f) * g_screen_w;
        float best_d2 = fov_px * fov_px;
        for (int i = 0; i < n; ++i) {
            uintptr_t actor_ptr = safe_read_uintptr(reinterpret_cast<void*>(data), i * 8);
            if (!sane_ptr(reinterpret_cast<void*>(actor_ptr))) continue;
            auto* actor = reinterpret_cast<ue4::UObject*>(actor_ptr);
            esp::Cat c = esp::target_category(actor, me_pawn);
            if (c == esp::Cat::Skip) continue;
            if (!cat_enabled(c)) continue;
            esp::Vec3 wp{};
            if (!esp::object_location(actor, wp)) continue;
            wp.z += g_aim_height;
            esp::Vec3 scr{};
            if (!esp::world_to_screen(wp, scr, g_screen_w, g_screen_h, cam_loc, cam_rot, cam_fov)) continue;
            if (!esp::on_screen(scr)) continue;
            if (g_aim_los && chain::g_los_fn &&
                !chain::has_los_cached(pc, actor, cam_loc.x, cam_loc.y, cam_loc.z)) continue;
            float dx = scr.x - cx, dy = scr.y - cy;
            float d2 = dx * dx + dy * dy;
            if (d2 < best_d2) { best_d2 = d2; r.pos = wp; r.actor = actor; r.found = true; }
        }
        return r;
    }
    inline ue4::UObject* g_aim_locked_actor = nullptr;
    inline void tick_aimbot() {
        if (!g_aim_on) { g_aim_locked_actor = nullptr; return; }
        bool hold_engaged = (GetAsyncKeyState(VK_RBUTTON) & 0x8000) != 0;
        bool should_run = g_aim_always ? g_aim_on : (hold_engaged || g_aim_on);
        if (!should_run) { g_aim_locked_actor = nullptr; return; }
        auto picked = pick_target();
        if (picked.found) {
            esp::Vec3 cam_loc{}; esp::Rot cam_rot{}; float cam_fov = 0.f;
            if (esp::read_camera(cam_loc, cam_rot, cam_fov)) {
                aim::Rot want = aim::look_at(cam_loc, picked.pos);
                silent::pitch = want.pitch; silent::yaw = want.yaw; silent::valid = true;
                bool is_new = (picked.actor != g_aim_locked_actor);
                if (is_new) {
                    g_aim_locked_actor = picked.actor;
                    if (g_aim_instant_lock && !g_aim_silent) { aim::set_control_rotation(want); return; }
                }
                if (g_aim_silent) return;
                aim::Rot cur = aim::get_control_rotation();
                aim::Rot applied;
                if (g_aim_snap) applied = want;
                else {
                    float t = (g_aim_smooth <= 1.f) ? 1.f : (1.f / g_aim_smooth);
                    applied = aim::lerp(cur, want, t);
                }
                aim::set_control_rotation(applied);
            }
        }
        else { silent::valid = false; g_aim_locked_actor = nullptr; }
    }

    inline void draw_esp() {
        if (!chain::world()) return;
        if (!g_esp_on && !g_aim_on) return;
        esp::Vec3 cam_loc{}; esp::Rot cam_rot{}; float cam_fov = 0.f;
        if (!esp::read_camera(cam_loc, cam_rot, cam_fov)) return;
        auto* me_pawn = chain::local_pawn();
        auto* pc = chain::local_pc();
        auto* arr = actors::get_actors();
        ImDrawList* dl = ImGui::GetBackgroundDrawList();
        const float cx = g_screen_w * 0.5f;
        const float cy = g_screen_h * 0.5f;
        if (g_aim_on && g_aim_show_fov_circle) {
            float fov_px = (g_aim_fov / 90.f) * g_screen_w;
            dl->AddCircle({ cx, cy }, fov_px, IM_COL32(255, 200, 80, 180), 64, 1.5f);
            dl->AddCircle({ cx, cy }, fov_px * 0.5f, IM_COL32(255, 200, 80, 90), 64, 1.0f);
        }
        if (!g_esp_on) return;
        if (arr) {
            uintptr_t arr_addr = reinterpret_cast<uintptr_t>(arr);
            int32_t n = safe_read_i32(reinterpret_cast<void*>(arr_addr), 8);
            uintptr_t data = safe_read_uintptr(reinterpret_cast<void*>(arr_addr), 0);
            const float max_d2 = g_esp_max_dist * g_esp_max_dist;
            for (int i = 0; i < n; ++i) {
                uintptr_t actor_ptr = safe_read_uintptr(reinterpret_cast<void*>(data), i * 8);
                if (!sane_ptr(reinterpret_cast<void*>(actor_ptr))) continue;
                auto* actor = reinterpret_cast<ue4::UObject*>(actor_ptr);
                esp::Cat c = esp::target_category(actor, me_pawn);
                if (c == esp::Cat::Skip) continue;
                if (!cat_enabled(c)) continue;
                esp::Vec3 wp{};
                if (!esp::object_location(actor, wp)) continue;
                float ddx = wp.x - cam_loc.x, ddy = wp.y - cam_loc.y, ddz = wp.z - cam_loc.z;
                float d2cm = (ddx * ddx + ddy * ddy + ddz * ddz) / 10000.f;
                if (d2cm > max_d2 || d2cm < 0.25f) continue;
                float dist = sqrtf(d2cm);
                esp::Vec3 head_w = wp; head_w.z += g_esp_head_off;
                esp::Vec3 feet_w = wp; feet_w.z += g_esp_feet_off;
                esp::Vec3 h_scr{}, f_scr{};
                if (!esp::world_to_screen(head_w, h_scr, g_screen_w, g_screen_h, cam_loc, cam_rot, cam_fov)) continue;
                if (!esp::world_to_screen(feet_w, f_scr, g_screen_w, g_screen_h, cam_loc, cam_rot, cam_fov)) continue;
                bool visible = true;
                if (chain::g_los_fn) visible = chain::has_los_cached(pc, actor, cam_loc.x, cam_loc.y, cam_loc.z);
                if (g_esp_visible && !visible) continue;
                float box_h = f_scr.y - h_scr.y;
                if (box_h < 4.f || box_h > g_esp_max_box_px) continue;
                float box_w = box_h * 0.5f;
                float x = h_scr.x - box_w * 0.5f;
                float y = h_scr.y;
                int cm = chams::g_mode;
                if (cm == (int)chams::Mode::Fill || cm == (int)chams::Mode::DepthFill) {
                    if (!chams::g_only_visible || visible) {
                        ImU32 col = visible ? chams::g_color_visible : chams::g_color_hidden;
                        dl->AddRectFilled({ x, y }, { x + box_w, y + box_h }, col);
                    }
                }
                if (cm == (int)chams::Mode::DepthFill || cm == (int)chams::Mode::DepthOnly) {
                    chams::apply_to(actor, visible);
                }
                if (g_esp_camera) {
                    dl->AddLine({ cx, cy }, { h_scr.x, h_scr.y }, IM_COL32(80, 200, 255, 160), 1.0f);
                    dl->AddCircle({ h_scr.x, h_scr.y }, 3.f, IM_COL32(80, 200, 255, 220), 8, 1.2f);
                }
                if (g_esp_box) {
                    ImU32 col = visible ? IM_COL32(0, 255, 120, 220) : IM_COL32(255, 60, 60, 220);
                    dl->AddRect({ x, y }, { x + box_w, y + box_h }, col, 0.f, 0, 1.5f);
                }
                if (g_esp_show_names) {
                    const std::string& cls = ue4::cached_name_of(actor);
                    if (!cls.empty()) {
                        ImVec2 sz = ImGui::CalcTextSize(cls.c_str());
                        ImVec2 np = { h_scr.x - sz.x * 0.5f, y - 32.f };
                        dl->AddText({ np.x + 1.f, np.y + 1.f }, IM_COL32(0, 0, 0, 220), cls.c_str());
                        dl->AddText(np, IM_COL32(255, 220, 100, 240), cls.c_str());
                    }
                }
                if (g_esp_distance) {
                    char lbl[64]{};
                    snprintf(lbl, sizeof(lbl), "%.0fm", dist);
                    dl->AddText({ x, y - 16.f }, IM_COL32(255, 255, 255, 220), lbl);
                }
            }
        }
        if (g_esp_self && me_pawn) {
            esp::Vec3 wp{};
            if (esp::object_location(me_pawn, wp)) {
                esp::Vec3 head_w = wp; head_w.z += g_esp_head_off;
                esp::Vec3 feet_w = wp; feet_w.z += g_esp_feet_off;
                esp::Vec3 h_scr{}, f_scr{};
                if (esp::world_to_screen(head_w, h_scr, g_screen_w, g_screen_h, cam_loc, cam_rot, cam_fov) &&
                    esp::world_to_screen(feet_w, f_scr, g_screen_w, g_screen_h, cam_loc, cam_rot, cam_fov)) {
                    float box_h = f_scr.y - h_scr.y;
                    float box_w = box_h * 0.5f;
                    float x = h_scr.x - box_w * 0.5f;
                    float y = h_scr.y;
                    dl->AddRect({ x, y }, { x + box_w, y + box_h }, IM_COL32(255, 80, 80, 255), 0.f, 0, 2.f);
                    dl->AddText({ x, y - 16.f }, IM_COL32(255, 80, 80, 255), "SELF");
                }
            }
        }
    }

    inline bool name_is_fire(const std::string& s) {
        return s == "Fire" || s == "ServerFire" || s == "FireWeapon" ||
            s == "ServerFireWeapon" || s == "StartFire" || s == "K2_Fire" ||
            s == "WeaponFire" || s == "FireShot" || s == "ServerFireShot";
    }
    inline void cache_fire_fns() {
        if (fn_get_camera_view && fn_fire) return;
        auto* cam = esp::camera_manager();
        if (cam && !fn_get_camera_view) fn_get_camera_view = chain::resolve_on_chain(cam, "GetCameraView");
        auto* pc = chain::local_pc();
        if (pc && !fn_fire) {
            uintptr_t cls = safe_read_uintptr(pc, off::UObject_Class);
            int depth = 0;
            while (cls && depth++ < 32) {
                uintptr_t child = safe_read_uintptr(reinterpret_cast<void*>(cls), off::UStruct_Children);
                int cc = 0;
                while (child && cc++ < 2000) {
                    ue4::FName n{};
                    if (!smemcpy(&n, reinterpret_cast<void*>(child + off::UObject_Name), sizeof(n))) break;
                    const std::string& s = ue4::name_to_string_cached(n);
                    if (!fn_fire && name_is_fire(s)) fn_fire = reinterpret_cast<ue4::UObject*>(child);
                    if (!fn_server_fire && s == "ServerFire") fn_server_fire = reinterpret_cast<ue4::UObject*>(child);
                    child = safe_read_uintptr(reinterpret_cast<void*>(child), off::UField_Next);
                }
                cls = safe_read_uintptr(reinterpret_cast<void*>(cls), off::UStruct_SuperStruct);
            }
        }
    }
    inline void __fastcall detour(ue4::UObject* obj, ue4::UObject* fn, void* params) {
        if (!o_process_event) return;
        if (fn) {
            ue4::FName fn_name{};
            if (smemcpy(&fn_name, reinterpret_cast<uint8_t*>(fn) + 0x18, sizeof(fn_name))) {
                if (stealth::is_target(fn_name.comparison_index)) return;
                if (features::g_debug_pe) {
                    const std::string& s = ue4::name_to_string_cached(fn_name);
                    if (!s.empty() && s.size() < 40) dbg("PE: %s", s.c_str());
                }
                if (recoil::intercept(fn, params)) return;
                bool is_fire = (fn == fn_fire || fn == fn_server_fire);
                if (is_fire && features::g_aim_silent && features::g_aim_on && silent::valid) {
                    auto* pc = chain::local_pc();
                    if (pc) {
                        aim::Rot saved = aim::get_control_rotation();
                        aim::Rot want{ silent::pitch, silent::yaw, 0.f };
                        aim::set_control_rotation(want);
                        o_process_event(obj, fn, params);
                        aim::set_control_rotation(saved);
                        return;
                    }
                }
                if (features::g_tp_on && fn == fn_get_camera_view && params) {
                    o_process_event(obj, fn, params);
                    uint8_t* base = reinterpret_cast<uint8_t*>(params);
                    float* loc = reinterpret_cast<float*>(base + 0x08);
                    float* rot = reinterpret_cast<float*>(base + 0x08 + 0x0C);
                    float pitch = rot[0] * 3.14159265f / 180.f;
                    float yaw = rot[1] * 3.14159265f / 180.f;
                    loc[0] -= cosf(pitch) * cosf(yaw) * features::g_tp_distance;
                    loc[1] -= cosf(pitch) * sinf(yaw) * features::g_tp_distance;
                    loc[2] -= sinf(pitch) * features::g_tp_distance;
                    loc[2] += features::g_tp_height;
                    return;
                }
            }
        }
        o_process_event(obj, fn, params);
    }
    inline void install() {
        void* addr = reinterpret_cast<void*>(ue4::base + off::ProcessEvent);
        MH_STATUS st = MH_CreateHook(addr, &detour, reinterpret_cast<LPVOID*>(&o_process_event));
        if (st != MH_OK) return;
        MH_EnableHook(addr);
    }

            ImGui_ImplWin32_WndProcHandler(h, m, w, l);
            if (g_imgui_active) return true;
        }
        return CallWindowProc(o_wndproc, h, m, w, l);
    }
    static HRESULT __stdcall hk_resize_buffers(IDXGISwapChain* sc, UINT bc, UINT w, UINT h, DXGI_FORMAT fmt, UINT flags) {
        if (g_ctx) { g_ctx->OMSetRenderTargets(0, nullptr, nullptr); g_ctx->Flush(); }
        if (g_rtv) { g_rtv->Release(); g_rtv = nullptr; }
        ImGui_ImplDX11_InvalidateDeviceObjects();
        HRESULT hr = o_resize_buffers(sc, bc, w, h, fmt, flags);
        if (SUCCEEDED(hr)) ImGui_ImplDX11_CreateDeviceObjects();
        return hr;
    }
    static void keybind_button(const char* id, int& key, bool& listen) {
        if (listen) { ImGui::Button("press any key...", ImVec2(150, 0)); return; }
        char label[64];
        const char* kname = nullptr;
        switch (key) {
        case VK_LBUTTON: kname = "Mouse1"; break;
        case VK_RBUTTON: kname = "Mouse2"; break;
        case VK_MBUTTON: kname = "Mouse3"; break;
        case VK_XBUTTON1: kname = "Mouse4"; break;
        case VK_XBUTTON2: kname = "Mouse5"; break;
        }
        if (kname) snprintf(label, sizeof(label), "%s  (change)", kname);
        else {
            UINT sc = MapVirtualKeyA(key, MAPVK_VK_TO_VSC);
            char keyname[32]{};
            if (sc && GetKeyNameTextA((LONG)(sc << 16), keyname, sizeof(keyname)))
                snprintf(label, sizeof(label), "%s  (change)", keyname);
            else snprintf(label, sizeof(label), "VK 0x%02X  (change)", key);
        }
        char final_label[96];
        snprintf(final_label, sizeof(final_label), "%s##%s", label, id);
        if (ImGui::Button(final_label, ImVec2(150, 0))) listen = true;
    }
    static LRESULT CALLBACK dummy_wndproc(HWND h, UINT m, WPARAM w, LPARAM l) { return DefWindowProc(h, m, w, l); }
    static HWND create_dummy_window() {
        static bool registered = false;
        const wchar_t* kClass = L"__fh_dummy_wnd";
        if (!registered) {
            WNDCLASSEXW wc{};
            wc.cbSize = sizeof(wc);
            wc.lpfnWndProc = dummy_wndproc;
            wc.hInstance = GetModuleHandleA(nullptr);
            wc.lpszClassName = kClass;
            if (!RegisterClassExW(&wc)) return nullptr;
            registered = true;
        }
        return CreateWindowExW(0, kClass, L"", WS_OVERLAPPEDWINDOW, 0, 0, 8, 8,
            nullptr, nullptr, GetModuleHandleA(nullptr), nullptr);
    }
    static void* try_create_swapchain_vtable(D3D_DRIVER_TYPE type, HWND hwnd, HRESULT& out_hr) {
        DXGI_SWAP_CHAIN_DESC sd{};
        sd.BufferCount = 1;
        sd.BufferDesc.Format = DXGI_FORMAT_R8G8B8A8_UNORM;
        sd.BufferDesc.RefreshRate.Numerator = 60; sd.BufferDesc.RefreshRate.Denominator = 1;
        sd.BufferUsage = DXGI_USAGE_RENDER_TARGET_OUTPUT;
        sd.OutputWindow = hwnd;
        sd.SampleDesc.Count = 1;
        sd.Windowed = TRUE;
        sd.SwapEffect = DXGI_SWAP_EFFECT_DISCARD;
        ID3D11Device* dev = nullptr;
        ID3D11DeviceContext* ctx = nullptr;
        IDXGISwapChain* sc = nullptr;
        D3D_FEATURE_LEVEL fl{};
        out_hr = D3D11CreateDeviceAndSwapChain(nullptr, type, nullptr, 0, nullptr, 0,
            D3D11_SDK_VERSION, &sd, &sc, &dev, &fl, &ctx);
        if (FAILED(out_hr) || !sc) {
            if (sc) sc->Release(); if (ctx) ctx->Release(); if (dev) dev->Release();
            return nullptr;
        }
        void** vtable = *reinterpret_cast<void***>(sc);
        void* result = (void*)vtable;
        sc->Release(); ctx->Release(); dev->Release();
        return result;
    }

                    if (ImGui::Button("Clear log")) features::g_log[0] = '\0';

                        ImGui::SameLine();

                    }
                    else g_copy_flash = false;
                    ImGui::Separator();
                    ImGui::Text("Screen: %dx%d", g_screen_w, g_screen_h);

                    ImGui::TextUnformatted(features::g_log);
                    ImGui::EndChild();
                    ImGui::EndTabItem();
                }

                ImGui::EndTabBar();
            }
            ImGui::End();
        }

        ImGui::Render();
        if (g_rtv) g_ctx->OMSetRenderTargets(1, &g_rtv, nullptr);
        ImGui_ImplDX11_RenderDrawData(ImGui::GetDrawData());
        return o_present(sc, sync, flags);
    }

    inline void install() {
        int waited = 0;
        while (!GetModuleHandleA("d3d11.dll")) {
            Sleep(500); waited += 500;
            if (waited > 120000) return;
        }
        HWND hwnd = create_dummy_window();
        if (!hwnd) return;
        struct Attempt { D3D_DRIVER_TYPE type; const char* name; };
        static const Attempt attempts[] = {
            { D3D_DRIVER_TYPE_NULL, "NULL" },
            { D3D_DRIVER_TYPE_HARDWARE, "HARDWARE" },
            { D3D_DRIVER_TYPE_WARP, "WARP" },
        };
        void** vtable = nullptr;
        for (const auto& a : attempts) {
            HRESULT hr = E_FAIL;
            vtable = reinterpret_cast<void**>(try_create_swapchain_vtable(a.type, hwnd, hr));
            if (vtable) break;
        }
        DestroyWindow(hwnd);
        if (!vtable) return;
        MH_STATUS st = MH_Initialize();
        if (st != MH_OK && st != MH_ERROR_ALREADY_INITIALIZED) return;
        MH_CreateHook(vtable[8], &hk_present, reinterpret_cast<LPVOID*>(&o_present));
        MH_CreateHook(vtable[13], &hk_resize_buffers, reinterpret_cast<LPVOID*>(&o_resize_buffers));
        MH_EnableHook(MH_ALL_HOOKS);
        pehook::install();
        speedhack::install();
    }

static DWORD WINAPI main_thread(LPVOID) {
    while (!GetModuleHandleA(nullptr)) Sleep(100);
    ue4::base = reinterpret_cast<uintptr_t>(GetModuleHandleA(nullptr));
    ue4::g_append_string = reinterpret_cast<ue4::AppendStringFn>(ue4::base + off::AppendString);
    ue4::g_process_event = reinterpret_cast<ue4::ProcessEventFn>(ue4::base + off::ProcessEvent);
    dbg("=== %s %s (%s) === base=0x%llX PE=%p",
        CHEAT_NAME, build_info::VERSION, build_info::BUILD_ID,
        ue4::base, ue4::g_process_event);
    features::config_load();
    sav_gen::get_username();
    sav_gen::g_folder = sav_gen::resolve_oar_save_folder();
    dbg("save folder resolved on inject: %s", sav_gen::g_folder.c_str());
    overlay::install();
    return 0;
}

BOOL APIENTRY DllMain(HMODULE h, DWORD reason, LPVOID) {
    if (reason == DLL_PROCESS_ATTACH) {
        DisableThreadLibraryCalls(h);
        open_console();
        dbg("=== DllMain attach ===");
        CreateThread(nullptr, 0, main_thread, nullptr, 0, nullptr);
    }
    return TRUE;
}
